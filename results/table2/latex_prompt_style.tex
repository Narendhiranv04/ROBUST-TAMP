\lstdefinestyle{promptstyle}{
  basicstyle=\ttfamily\scriptsize, columns=fullflexible, keepspaces=true,
  breaklines=true, breakatwhitespace=true, showstringspaces=false, upquote=true,
  breakindent=1.2em, postbreak=\mbox{\textcolor{pbGray}{$\hookrightarrow$}\space},
  moredelim=[l][\color{pbBlue}\bfseries]{\#\#},
  emph={FINAL,ACTIONS,BLOCKS,BLOCK,END,NO_ACTIONS}, emphstyle=\color{pbOrange}\bfseries,
  emph={[2]Preconditions,Effects,urgency,insert,objects,reason}, emphstyle={[2]\color{pbTeal}\bfseries},
}
